# SCAR-L1 Implementation Plan

> **For agentic workers:** REQUIRED: Use superpowers:subagent-driven-development (if subagents available) or superpowers:executing-plans to implement this plan. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build SCAR-L1, a state-certified L1 reuse layer for TraceCAG that accepts safe near-hit patch/reuse cases while rejecting unsafe state drift.

**Architecture:** Add a focused `l1_state_cache.py` module for signatures, certificates, hard rejects, risk scoring, candidate ranking, and decisions. Keep `nodes_v2.py` as orchestration: L0 exact first, SCAR-L1 for L1 candidate evaluation, L2 full path otherwise.

**Tech Stack:** Python 3, TypedDict/dataclasses, pytest, pytest-asyncio, existing TraceCAG in-memory/Redis cache structures.

```python
from __future__ import annotations

from dataclasses import dataclass, field
import time


@dataclass(frozen=True)
class L1Request:
    query_norm: str
    intent: str
    level: str
    profile_epoch: int
    session_turn: int
    concepts: set[str] = field(default_factory=set)
    entities: set[str] = field(default_factory=set)
    answer_target: str = ""
    relation_hints: set[str] = field(default_factory=set)
    evidence_hash: str = ""


@dataclass(frozen=True)
class L1Candidate:
    cache_key: str
    query_norm: str
    intent: str
    level: str
    profile_epoch: int
    session_turn: int
    concepts: set[str] = field(default_factory=set)
    entities: set[str] = field(default_factory=set)
    answer_target: str = ""
    relation_hints: set[str] = field(default_factory=set)
    evidence_hash: str = ""
    created_at: float = 0.0
    ttl: int = 3600


@dataclass(frozen=True)
class L1Decision:
    decision: str
    risk: float
    rank_score: float
    reasons: tuple[str, ...]
    safe_to_reuse: bool


def _jaccard(a: set[str], b: set[str]) -> float:
    if not a and not b:
        return 1.0
    return len(a & b) / max(len(a | b), 1)


def decide_l1_reuse(
    request: L1Request,
    candidate: L1Candidate,
    *,
    now: float | None = None,
    tau_reuse: float = 0.25,
    tau_patch: float = 0.55,
    concept_floor: float = 0.50,
) -> L1Decision:
    now = time.monotonic() if now is None else now
    reasons: list[str] = []

    if request.level != candidate.level:
        reasons.append("level_mismatch")
    if request.profile_epoch != candidate.profile_epoch:
        reasons.append("profile_epoch_mismatch")
    if request.intent and candidate.intent and request.intent != candidate.intent:
        reasons.append("intent_mismatch")
    if request.answer_target and candidate.answer_target and request.answer_target != candidate.answer_target:
        reasons.append("answer_target_mismatch")
    if request.evidence_hash and candidate.evidence_hash and request.evidence_hash != candidate.evidence_hash:
        reasons.append("evidence_mismatch")
    if request.relation_hints and candidate.relation_hints and request.relation_hints != candidate.relation_hints:
        reasons.append("relation_mismatch")

    concept_overlap = _jaccard(request.concepts, candidate.concepts)
    if concept_overlap < concept_floor:
        reasons.append("concept_overlap_below_floor")

    age_ratio = min(max((now - candidate.created_at) / max(candidate.ttl, 1), 0.0), 1.0)
    surface_same = request.query_norm == candidate.query_norm
    entity_overlap = _jaccard(request.entities, candidate.entities)
    relation_overlap = _jaccard(request.relation_hints, candidate.relation_hints)

    risk = (
        0.30 * (0.0 if request.intent == candidate.intent else 1.0)
        + 0.25 * (1.0 - concept_overlap)
        + 0.20 * (1.0 - entity_overlap)
        + 0.15 * (1.0 - relation_overlap)
        + 0.10 * age_ratio
    )
    risk = max(0.0, min(1.0, risk))
    rank_score = (1.0 - risk) + (0.3 * concept_overlap) + (0.2 * entity_overlap)

    if reasons:
        return L1Decision("full", risk, rank_score, tuple(reasons), False)
    if surface_same and risk <= tau_reuse:
        return L1Decision("reuse", risk, rank_score, ("exact_or_normalized_match",), True)
    if risk <= tau_patch:
        return L1Decision("patch", risk, rank_score, ("state_compatible_near_hit",), True)
    return L1Decision("full", risk, rank_score, ("risk_above_patch_threshold",), False)
```

```python
def _answer_target_hint(user_input: str) -> str:
    text = (user_input or "").lower()
    if any(token in text for token in ["who", "founder", "mother", "father", "director"]):
        return "person"
    if any(token in text for token in ["when", "year", "date"]):
        return "time"
    if any(token in text for token in ["where", "country", "city", "place"]):
        return "place"
    if any(token in text for token in ["how many", "times", "number"]):
        return "number"
    return "answer"


def _relation_hints(user_input: str) -> set[str]:
    text = (user_input or "").lower()
    hints: set[str] = set()
    relation_tokens = {
        "founder": ["founder", "founded", "created"],
        "birth": ["born", "birth"],
        "director": ["director", "directed"],
        "family": ["mother", "father", "wife", "brother", "grandfather"],
        "time": ["year", "date", "came first", "released"],
        "location": ["where", "country", "city", "place"],
    }
    for hint, tokens in relation_tokens.items():
        if any(token in text for token in tokens):
            hints.add(hint)
    return hints
```
